const StatusBadge = ({ status }) => {
  const styles = {
    completed: "bg-emerald-500/15 text-emerald-400 border border-emerald-500/30",
    running: "bg-blue-500/15 text-blue-400 border border-blue-500/30",
    scheduled: "bg-amber-500/15 text-amber-400 border border-amber-500/30",
    draft: "bg-zinc-500/15 text-zinc-400 border border-zinc-500/30",
    open: "bg-blue-500/15 text-blue-400 border border-blue-500/30",
    bot: "bg-violet-500/15 text-violet-400 border border-violet-500/30",
    assigned: "bg-amber-500/15 text-amber-400 border border-amber-500/30",
    closed: "bg-zinc-500/15 text-zinc-400 border border-zinc-500/30",
  };
  return (
    <span className={`text-xs font-medium px-2 py-0.5 rounded-full ${styles[status] || styles.draft}`}>
      {status.charAt(0).toUpperCase() + status.slice(1)}
    </span>
  );
};

const MetricCard = ({ label, value, sub, trend, color = "indigo" }) => {
  const colors = {
    indigo: "from-indigo-600/20 to-indigo-900/5 border-indigo-500/20",
    emerald: "from-emerald-600/20 to-emerald-900/5 border-emerald-500/20",
    amber: "from-amber-600/20 to-amber-900/5 border-amber-500/20",
    blue: "from-blue-600/20 to-blue-900/5 border-blue-500/20",
  };
  const dotColors = { indigo: "bg-indigo-400", emerald: "bg-emerald-400", amber: "bg-amber-400", blue: "bg-blue-400" };
  return (
    <div className={`bg-gradient-to-br ${colors[color]} border rounded-2xl p-5 relative overflow-hidden`}>
      <div className={`absolute top-3 right-3 w-2 h-2 rounded-full ${dotColors[color]} animate-pulse`} />
      <p className="text-xs text-zinc-500 font-medium uppercase tracking-widest mb-2">{label}</p>
      <p className="text-3xl font-bold text-white tracking-tight">{value}</p>
      {sub && <p className="text-sm text-zinc-400 mt-1">{sub}</p>}
      {trend && <p className={`text-xs mt-2 font-medium ${trend > 0 ? 'text-emerald-400' : 'text-red-400'}`}>
        {trend > 0 ? '↑' : '↓'} {Math.abs(trend)}% vs last week
      </p>}
    </div>
  );
};
